<?php
if(isset($_POST["str"]))
{
    $str = $_POST["str"];
    $upper = 0;
    $lower = 0;

    for($i=0; $i<strlen($str); $i++)
    {
        if($str[$i]>='A' && $str[$i]<='Z')
            $upper++;

        elseif($str[$i]>='a' && $str[$i]<='z')
            $lower++;
    }

    echo "<br>Original String: ".$str;
    echo "<br>Uppercase Letters: ".$upper;
    echo "<br>Lowercase Letters: ".$lower;

    echo "<br>Reversed Words: ";
    $words = explode(" ", $str);

    foreach($words as $word)
    {
        echo strrev($word)." ";
    }
}
