<?php

// Check whether the cookie already exists
if (isset($_COOKIE["last_visit"])) {
    $last_visit = $_COOKIE["last_visit"];
} else {
    $last_visit = "This is your first visit!";
}

// Store the current date and time in the cookie
$current_time = date("d-m-Y h:i:s A");

setcookie("last_visit", $current_time, time() + (86400 * 30));

?>

<!DOCTYPE html>
<html>
<head>
    <title>Last Visit</title>
</head>

<body>

    <h1>Welcome to My Web Page</h1>

    <h2>Last Visited:</h2>

    <p>
        <?php echo $last_visit; ?>
    </p>

</body>
</html>
